-- Assumptions: caps out at 255 characters
-- event title and location
-- participant name and email
-- does not check for registration time. left to main

CREATE DATABASE assignment1;

USE assignment1;

--Create Tables
CREATE TABLE events (
    id int AUTO_INCREMENT PRIMARY KEY,
    name varchar(255) NOT NULL,
    date DATE NOT NULL,
    time TIME NOT NULL,
    capicity int UNSIGNED NOT NULL,
    location varchar(255) NOT NULL 
);

CREATE TABLE participants (
    userID int AUTO_INCREMENT PRIMARY KEY,
    name varchar(255) NOT NULL,
    email varchar(255) UNIQUE NOT NULL
);

CREATE TABLE registrations (
    registrationID int AUTO_INCREMENT PRIMARY KEY,
    eventID int NOT NULL,
    userID int NOT NULL,
    CONSTRAINT uniqueRegistration UNIQUE (eventID, userID),
    FOREIGN KEY (eventID) REFERENCES events(id),
    FOREIGN KEY (userID) REFERENCES participants(userID),
    currentTime DATETIME DEFAULT CURRENT_TIMESTAMP
);
